104static Node *parse_prefix(void) {
105 while (*pp && isspace((unsigned char)*pp)) pp++; /* skip spaces */
106
107 if (*pp == '\0') {
108 fprintf(stderr, "Error: unexpected end of prefix expression\n");
109 exit(1);
110 }
111
112 /* binary connective: read left subtree, then right subtree */
113 if (is_binary_op(*pp)) {
114 char op[2] = { *pp, '\0' };
115 pp++;
116 Node *left = parse_prefix();
117 Node *right = parse_prefix();
118 return new_node(op, left, right);
119 }
120
121 /* unary negation: one child (stored on the left) */
122 if (*pp == '~') {
123 pp++;
124 Node *child = parse_prefix();
125 return new_node("~", child, NULL);
126 }
127
128 /* atom: letters/digits */
129 char buf[32];
130 int i = 0;
131 while (isalnum((unsigned char)*pp) && i < 31) buf[i++] = *pp++;
132 buf[i] = '\0';
133 if (i == 0) {
134 fprintf(stderr, "Error: invalid symbol '%c'\n", *pp);
135 exit(1);
136 }
137 return new_node(buf, NULL, NULL);
138}

200static int tokenize_formula(const char *str, Token *tokens, int max_tokens) {
201 int count = 0;
202 const char *s = str;
203 while (*s && count < max_tokens) {
204 while (*s && isspace((unsigned char)*s)) s++;
205 if (!*s) break;
206 if (*s == '(' || *s == ')' || is_connective(*s)) {
207 tokens[count].text[0] = *s++;
208 tokens[count].text[1] = '\0';
209 count++;
210 } else if (isalnum((unsigned char)*s)) {
211 int j = 0;
212 while (isalnum((unsigned char)*s) && j < 31) {
213 tokens[count].text[j++] = *s++;
214 }
215 tokens[count].text[j] = '\0';
216 count++;
217 } else {
218 s++;
219 }
220 }
221 return count;
222}

224static void infix_to_prefix_helper(const char *infix, char *prefix_out) {
225 Token *tokens = (Token *)malloc(MAX_TOKENS * sizeof(Token));
226 Token *out = (Token *)malloc(MAX_TOKENS * sizeof(Token));
227 char *op_stack = (char *)malloc(MAX_TOKENS * sizeof(char));
228 if (!tokens || !out || !op_stack) {
229 fprintf(stderr, "Out of memory in conversion\n");
230 exit(1);
231 }
232
233 int n = tokenize_formula(infix, tokens, MAX_TOKENS);
234 int top = 0;
235 int out_count = 0;
236
237 for (int i = n - 1; i >= 0; i--) {
238 char *t = tokens[i].text;
239 if (strcmp(t, ")") == 0) {
240 op_stack[top++] = ')';
241 } else if (strcmp(t, "(") == 0) {
242 while (top > 0 && op_stack[top - 1] != ')') {
243 out[out_count].text[0] = op_stack[--top];
244 out[out_count].text[1] = '\0';
245 out_count++;
246 }
247 if (top > 0 && op_stack[top - 1] == ')') {
248 top--;
249 }
250 } else if (is_connective(t[0]) && t[1] == '\0') {
251 char op = t[0];
252 while (top > 0 && op_stack[top - 1] != ')' &&
253 connective_precedence(op_stack[top - 1]) > connective_precedence(op)) {
254 out[out_count].text[0] = op_stack[--top];
255 out[out_count].text[1] = '\0';
256 out_count++;
257 }
258 op_stack[top++] = op;
259 } else {
260 strncpy(out[out_count].text, t, sizeof(out[out_count].text) - 1);
261 out[out_count].text[sizeof(out[out_count].text) - 1] = '\0';
262 out_count++;
263 }
264 }
265
266 while (top > 0) {
267 if (op_stack[top - 1] != ')') {
268 out[out_count].text[0] = op_stack[top - 1];
269 out[out_count].text[1] = '\0';
270 out_count++;
271 }
272 top--;
273 }
274
275 prefix_out[0] = '\0';
276 size_t cur_len = 0;
277 for (int i = out_count - 1; i >= 0; i--) {
278 size_t tlen = strlen(out[i].text);
279 if (cur_len > 0) {
280 prefix_out[cur_len++] = ' ';
281 prefix_out[cur_len] = '\0';
282 }
283 strcat(prefix_out, out[i].text);
284 cur_len += tlen;
285 }
286
287 free(tokens);
288 free(out);
289 free(op_stack);
290}

313int read_input_cnf_as_prefix(const char *filename, char *prefix_out, size_t max_len) {
314 FILE *fp = fopen(filename, "r");
315 if (!fp) {
316 perror("Error opening CNF file");
317 return -1;
318 }
319
320 char *infix = (char *)malloc(max_len);
321 if (!infix) { fclose(fp); return -1; }
322 infix[0] = '\0';
323
324 char line[4096];
325 int num_vars = 0, num_clauses = 0;
326 char current_clause[8192] = "";
327 int lits_in_clause = 0;
328
329 while (fgets(line, sizeof(line), fp)) {
330 char *ptr = line;
331 while (*ptr && isspace((unsigned char)*ptr)) ptr++;
332 if (*ptr == '\0' || *ptr == 'c') continue;
333 if (*ptr == 'p') {
334 sscanf(ptr, "p cnf %d %d", &num_vars, &num_clauses);
335 continue;
336 }
337
338 char *token = strtok(ptr, " \t\r\n");
339 while (token) {
340 int lit = atoi(token);
341 if (lit == 0) {
342 if (lits_in_clause > 0) {
343 if (infix[0] == '\0') {
344 strncpy(infix, current_clause, max_len - 1);
345 infix[max_len - 1] = '\0';
346 } else {
347 size_t needed = strlen(infix) + strlen(current_clause) + 6;
348 if (needed < max_len) {
349 char *temp = (char *)malloc(max_len);
350 if (temp) {
351 snprintf(temp, max_len, "(%s*%s)", infix, current_clause);
352 strncpy(infix, temp, max_len - 1);
353 infix[max_len - 1] = '\0';
354 free(temp);
355 }
356 } else {
357 fclose(fp);
358 infix_to_prefix_helper(infix, prefix_out);
359 free(infix);
360 return 0;
361 }
362 }
363 current_clause[0] = '\0';
364 lits_in_clause = 0;
365 }
366 } else {
367 char lit_str[64];

371 snprintf(lit_str, sizeof(lit_str), "x%d", lit);

376 current_clause[sizeof(current_clause) - 1] = '\0';
377 } else {

380 strncpy(current_clause, temp, sizeof(current_clause) - 1);
381 current_clause[sizeof(current_clause) - 1] = '\0';

383 lits_in_clause++;

387 }
