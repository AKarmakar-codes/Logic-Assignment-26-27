101static Node *parse_prefix(void) {
102 while (*pp && isspace((unsigned char)*pp)) pp++; /* skip spaces */
103
104 if (*pp == '\0') {
105 fprintf(stderr, "Error: unexpected end of prefix expression\n");
106 exit(1);
107 }
108
109 /* binary connective: read left subtree, then right subtree */
110 if (is_binary_op(*pp)) {
111 char op[2] = { *pp, '\0' };
112 pp++;
113 Node *left = parse_prefix();
114 Node *right = parse_prefix();
115 return new_node(op, left, right);
116 }
117
118 /* unary negation: one child (stored on the left) */
119 if (*pp == '~') {
120 pp++;
121 Node *child = parse_prefix();
122 return new_node("~", child, NULL);
123 }
124
125 /* atom: letters/digits */
126 char buf[32];
127 int i = 0;
128 while (isalnum((unsigned char)*pp) && i < 31) buf[i++] = *pp++;
129 buf[i] = '\0';
130 if (i == 0) {
131 fprintf(stderr, "Error: invalid symbol '%c'\n", *pp);
132 exit(1);
133 }
134 return new_node(buf, NULL, NULL);
135}
