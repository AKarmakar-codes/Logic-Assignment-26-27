181int collect_atoms(const Node *n, AtomTable *t) {
182 if (!n) return 0;
183 if (is_atom(n)) {
184 for (int i = 0; i < t->count; i++)
185 if (strcmp(t->names[i], n->label) == 0) return 0; /* already known */
186 if (t->count == MAX_ATOMS) return -1;
187 strcpy(t->names[t->count++], n->label);
188 return 0;
189 }
190 if (collect_atoms(n->left, t) < 0) return -1;
191 return collect_atoms(n->right, t);
192}

221int evaluate(const Node *n, const AtomTable *t, const int *val) {
222 if (is_atom(n))
223 return val[atom_index(t, n->label)];
224
225 if (is_negation(n))
226 return !evaluate(n->left, t, val);
227
228 int a = evaluate(n->left, t, val); /* children first ... */
229 int b = evaluate(n->right, t, val); /* ... then the connective */
230
231 switch (n->label[0]) {
232 case '*': return a && b; /* AND */
233 case '+': return a || b; /* OR */
234 case '>': return !a || b; /* implication */
235 default:
236 fprintf(stderr, "Error: unknown connective '%s'\n", n->label);
237 exit(1);
238 }
239}

252void print_truth_table(const Node *root, const AtomTable *t, const char *formula) {
253 int k = t->count;
254 long rows = 1L << k;
255 long true_rows = 0;
256 int val[MAX_ATOMS];
257
258 /* header */
259 for (int i = 0; i < k; i++) printf(" %-5s|", t->names[i]);
260 printf(" %s\n", formula);
261 for (int i = 0; i < k; i++) printf("-------+");
262 printf("-------\n");
263
264 for (long r = 0; r < rows; r++) {
265 for (int i = 0; i < k; i++)
266 val[i] = !((r >> (k - 1 - i)) & 1); /* first row: all true */
267
268 int result = evaluate(root, t, val);
269 true_rows += result;
270
271 for (int i = 0; i < k; i++) printf(" %-5c|", val[i] ? 'T' : 'F');
272 printf(" %c\n", result ? 'T' : 'F');
273 }
274
275 printf("\nRows evaluating to true: %ld of %ld\n", true_rows, rows);
276 if (true_rows == rows) printf("The formula is a TAUTOLOGY (valid).\n");
277 else if (true_rows == 0) printf("The formula is a CONTRADICTION (unsatisfiable).\n");
278 else printf("The formula is SATISFIABLE but not valid.\n");
279}
