252static void stack_push(Stack *s, const Node *n, int state) {
253 if (s->top == s->cap) {
254 s->cap *= 2;
255 s->data = (Frame *)realloc(s->data, s->cap * sizeof(Frame));
256 if (!s->data) { fprintf(stderr, "Out of memory\n"); exit(1); }
257 }
258 s->data[s->top].node = n;
259 s->data[s->top].state = state;
260 s->top++;
261}

294void inorder_iter(const Node *root) {
295 if (!root) return;
296 Stack s;
297 stack_init(&s);
298 stack_push(&s, root, 0);

300 while (!stack_empty(&s)) {

302 const Node *n = f->node;
303
304 if (is_atom(n)) {
305 printf("%s", n->label);
306 s.top--;
307 } else if (is_negation(n)) {

312 } else {
313 s.top--;

409int read_input_cnf(const char *filename, char *buffer, size_t max_len) {
410 FILE *fp = fopen(filename, "r");
411 if (!fp) {
412 perror("Error opening CNF file");

414 }
415
416 char line[4096];
417 int num_vars = 0, num_clauses = 0;
418 buffer[0] = '\0';
419
420 char current_clause[8192] = "";
421 int lits_in_clause = 0;
422
423 while (fgets(line, sizeof(line), fp)) {
424 char *ptr = line;
425 while (*ptr && isspace((unsigned char)*ptr)) ptr++;
426 if (*ptr == '\0' || *ptr == 'c') continue;
427 if (*ptr == 'p') {
428 sscanf(ptr, "p cnf %d %d", &num_vars, &num_clauses);
429 continue;
430 }
431
432 char *token = strtok(ptr, " \t\r\n");
433 while (token) {
434 int lit = atoi(token);
435 if (lit == 0) {
436 if (lits_in_clause > 0) {
437 if (buffer[0] == '\0') {
438 strncpy(buffer, current_clause, max_len - 1);
439 buffer[max_len - 1] = '\0';
440 } else {
441 size_t needed = strlen(buffer) + strlen(current_clause) + 6;
442 if (needed < max_len) {
443 char *temp = (char *)malloc(max_len);
444 if (temp) {
445 snprintf(temp, max_len, "(%s*%s)", buffer, current_clause);
446 strncpy(buffer, temp, max_len - 1);
447 buffer[max_len - 1] = '\0';
448 free(temp);
449 }
450 } else {
451 fclose(fp);
452 return 0;
453 }
454 }
455 current_clause[0] = '\0';
456 lits_in_clause = 0;
457 }
458 } else {
459 char lit_str[64];
