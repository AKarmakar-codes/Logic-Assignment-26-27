252static void stack_push(Stack *s, const Node *n, int state) {
253 if (s->top == s->cap) {

294void inorder_iter(const Node *root) {
295 if (!root) return;
296 Stack s;
297 stack_init(&s);
298 stack_push(&s, root, 0);
299
300 while (!stack_empty(&s)) {
301 Frame *f = &s.data[s.top - 1];
302 const Node *n = f->node;
303
304 if (is_atom(n)) {
305 printf("%s", n->label);
306 s.top--;
307 } else if (is_negation(n)) {
308 if (f->state == 0) {

312 } else {
313 s.top--;
314 }
315 } else { /* binary connective */
