86Node *parse_formula(void) {
87 skip_spaces();
88
89 if (*p == '(') {
90 p++; /* consume '(' */
91 Node *left = parse_formula();
92 skip_spaces();
93 if (*p == ')') { /* e.g. (p) or (~p) */
94 p++;
95 return left;
96 }
97 char op[2] = { *p, '\0' }; /* + * > */
98 p++;
99 Node *right = parse_formula();
100 skip_spaces();
101 if (*p == ')') p++; /* consume ')' */
102 return new_node(op, left, right);
103 }
104
105 if (*p == '~') {
106 p++;
107 Node *child = parse_formula();
108 return new_node("~", child, NULL);
109 }

111 /* atom: letters/digits, e.g. p, q1, x */
112 char buf[32];
113 int i = 0;
114 while (isalnum((unsigned char)*p) && i < 31) buf[i++] = *p++;
115 buf[i] = '\0';
116 return new_node(buf, NULL, NULL);

190static int connective_precedence(char c) {
191 switch (c) {
192 case '~': return 4;
193 case '*': return 3;
194 case '+': return 2;
195 case '>': return 1;
196 default: return 0;
197 }

208static int tokenize_formula(const char *str, Token *tokens, int max_tokens) {
209 int count = 0;
210 const char *s = str;
211 while (*s && count < max_tokens) {
212 while (*s && isspace((unsigned char)*s)) s++;
213 if (!*s) break;
214 if (*s == '(' || *s == ')' || is_connective(*s)) {
215 tokens[count].text[0] = *s++;
216 tokens[count].text[1] = '\0';
217 count++;
218 } else if (isalnum((unsigned char)*s)) {
219 int j = 0;
220 while (isalnum((unsigned char)*s) && j < 31) {
221 tokens[count].text[j++] = *s++;
222 }
223 tokens[count].text[j] = '\0';
224 count++;
225 } else {
226 s++; /* Ignore any extraneous characters */
227 }
228 }
229 return count;
230}

246void infix_to_prefix_stack(const char *infix, char *prefix_out) {
247 Token *tokens = (Token *)malloc(MAX_TOKENS * sizeof(Token));
248 Token *out = (Token *)malloc(MAX_TOKENS * sizeof(Token));
249 char *op_stack = (char *)malloc(MAX_TOKENS * sizeof(char));
250 if (!tokens || !out || !op_stack) {
251 fprintf(stderr, "Error: out of memory in infix_to_prefix_stack\n");
252 exit(1);
253 }
254
255 int n = tokenize_formula(infix, tokens, MAX_TOKENS);
256 int top = 0;
257 int out_count = 0;
258
259 for (int i = n - 1; i >= 0; i--) {
260 char *t = tokens[i].text;
261 if (strcmp(t, ")") == 0) {
262 op_stack[top++] = ')';
263 } else if (strcmp(t, "(") == 0) {
264 while (top > 0 && op_stack[top - 1] != ')') {
265 out[out_count].text[0] = op_stack[--top];
266 out[out_count].text[1] = '\0';
267 out_count++;
268 }
269 if (top > 0 && op_stack[top - 1] == ')') {
270 top--; /* Pop matching ')' */
271 }
272 } else if (is_connective(t[0]) && t[1] == '\0') {
273 char op = t[0];
274 while (top > 0 && op_stack[top - 1] != ')' &&
275 connective_precedence(op_stack[top - 1]) > connective_precedence(op)) {
276 out[out_count].text[0] = op_stack[--top];
277 out[out_count].text[1] = '\0';
278 out_count++;
279 }
280 op_stack[top++] = op;
281 } else {
282 /* Atom / Operand */
283 strncpy(out[out_count].text, t, sizeof(out[out_count].text) - 1);
284 out[out_count].text[sizeof(out[out_count].text) - 1] = '\0';
285 out_count++;
286 }
287 }
288
289 /* Pop any remaining operators from the stack */
290 while (top > 0) {
291 if (op_stack[top - 1] != ')') {
292 out[out_count].text[0] = op_stack[top - 1];
293 out[out_count].text[1] = '\0';
294 out_count++;
295 }
296 top--;
297 }
298
299 /* Reverse output tokens to produce prefix ordering */
300 prefix_out[0] = '\0';
301 size_t cur_len = 0;
302 for (int i = out_count - 1; i >= 0; i--) {
303 size_t tlen = strlen(out[i].text);
304 if (cur_len > 0) {
305 prefix_out[cur_len++] = ' ';
306 prefix_out[cur_len] = '\0';
307 }
308 strcat(prefix_out, out[i].text);
309 cur_len += tlen;
310 }
311
312 free(tokens);
313 free(out);
314 free(op_stack);
315}

366int read_input_cnf(const char *filename, char *buffer, size_t max_len) {
367 FILE *fp = fopen(filename, "r");
368 if (!fp) {
369 perror("Error opening CNF file");
370 return -1;
371 }
372
373 char line[4096];
374 int num_vars = 0, num_clauses = 0;
375 buffer[0] = '\0';
376
377 char current_clause[8192] = "";
378 int lits_in_clause = 0;
379
380 while (fgets(line, sizeof(line), fp)) {
381 char *ptr = line;
382 while (*ptr && isspace((unsigned char)*ptr)) ptr++;
383 if (*ptr == '\0' || *ptr == 'c') continue; /* Comment / blank */
384 if (*ptr == 'p') {
385 sscanf(ptr, "p cnf %d %d", &num_vars, &num_clauses);
386 continue;
387 }
388
389 char *token = strtok(ptr, " \t\r\n");
390 while (token) {
391 int lit = atoi(token);
392 if (lit == 0) {
393 /* End of clause */
394 if (lits_in_clause > 0) {
395 if (buffer[0] == '\0') {
396 strncpy(buffer, current_clause, max_len - 1);
397 buffer[max_len - 1] = '\0';
398 } else {
399 size_t needed = strlen(buffer) + strlen(current_clause) + 6;
400 if (needed < max_len) {
401 char *temp = (char *)malloc(max_len);
402 if (temp) {
403 snprintf(temp, max_len, "(%s*%s)", buffer, current_clause);
404 strncpy(buffer, temp, max_len - 1);
405 buffer[max_len - 1] = '\0';
406 free(temp);
407 }
408 } else {
409 /* Buffer limit reached: return safely */
410 fclose(fp);
411 return 0;
412 }
413 }
414 current_clause[0] = '\0';
415 lits_in_clause = 0;
416 }
417 } else {
418 /* Literal representation */
419 char lit_str[64];
420 if (lit < 0) {
421 snprintf(lit_str, sizeof(lit_str), "(~x%d)", -lit);
422 } else {
423 snprintf(lit_str, sizeof(lit_str), "x%d", lit);
424 }
425
426 if (lits_in_clause == 0) {
427 strncpy(current_clause, lit_str, sizeof(current_clause) - 1);
428 current_clause[sizeof(current_clause) - 1] = '\0';
429 } else {
430 char temp[8192];
431 snprintf(temp, sizeof(temp), "(%s+%s)", current_clause, lit_str);
432 strncpy(current_clause, temp, sizeof(current_clause) - 1);
433 current_clause[sizeof(current_clause) - 1] = '\0';
434 }
435 lits_in_clause++;
436 }
437 token = strtok(NULL, " \t\r\n");
438 }
439 }
440
441 fclose(fp);
442 return 0;
443}
