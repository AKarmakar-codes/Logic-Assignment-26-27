163int height_iter(const Node *root) {
164 if (!root) return -1;
165
166 int cap = 64, head = 0, tail = 0;
167 const Node **queue = (const Node **)malloc(cap * sizeof(const Node *));
168 if (!queue) { fprintf(stderr, "Out of memory\n"); exit(1); }
169
170 queue[tail++] = root;
171 int height = -1;
172
173 while (head < tail) {
174 int level_size = tail - head; /* nodes on the current level */
175 height++;
176 for (int i = 0; i < level_size; i++) {
177 const Node *n = queue[head++];
178 int need = (n->left ? 1 : 0) + (n->right ? 1 : 0);
179 if (tail + need > cap) { /* grow the queue if needed */
180 cap *= 2;
181 queue = (const Node **)realloc(queue, cap * sizeof(const Node *));
182 if (!queue) { fprintf(stderr, "Out of memory\n"); exit(1); }
183 }
184 if (n->left) queue[tail++] = n->left;
185 if (n->right) queue[tail++] = n->right;
186 }
187 }
188 free(queue);
189 return height;
190}
