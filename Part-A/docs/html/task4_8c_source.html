163int height_iter(const Node *root) {
164 if (!root) return -1;
165
166 int cap = 64, head = 0, tail = 0;
167 const Node **queue = (const Node **)malloc(cap * sizeof(const Node *));
168 if (!queue) { fprintf(stderr, "Out of memory\n"); exit(1); }
169
170 queue[tail++] = root;
171 int height = -1;
172
173 while (head < tail) {
174 int level_size = tail - head; /* nodes on the current level */
175 height++;
176 for (int i = 0; i < level_size; i++) {
177 const Node *n = queue[head++];
178 int need = (n->left ? 1 : 0) + (n->right ? 1 : 0);
179 if (tail + need > cap) { /* grow the queue if needed */
180 cap *= 2;
181 queue = (const Node **)realloc(queue, cap * sizeof(const Node *));
182 if (!queue) { fprintf(stderr, "Out of memory\n"); exit(1); }
183 }
184 if (n->left) queue[tail++] = n->left;
185 if (n->right) queue[tail++] = n->right;
186 }
187 }
188 free(queue);
189 return height;
190}

219int read_input_cnf(const char *filename, char *buffer, size_t max_len) {
220 FILE *fp = fopen(filename, "r");
221 if (!fp) {
222 perror("Error opening CNF file");

224 }
225
226 char line[4096];
227 int num_vars = 0, num_clauses = 0;
228 buffer[0] = '\0';
229
230 char current_clause[8192] = "";
231 int lits_in_clause = 0;
232
233 while (fgets(line, sizeof(line), fp)) {
234 char *ptr = line;
235 while (*ptr && isspace((unsigned char)*ptr)) ptr++;
236 if (*ptr == '\0' || *ptr == 'c') continue;
237 if (*ptr == 'p') {
238 sscanf(ptr, "p cnf %d %d", &num_vars, &num_clauses);
239 continue;
240 }
241
242 char *token = strtok(ptr, " \t\r\n");
243 while (token) {
244 int lit = atoi(token);
245 if (lit == 0) {
246 if (lits_in_clause > 0) {
247 if (buffer[0] == '\0') {
248 strncpy(buffer, current_clause, max_len - 1);
249 buffer[max_len - 1] = '\0';
250 } else {
251 size_t needed = strlen(buffer) + strlen(current_clause) + 6;
252 if (needed < max_len) {
253 char *temp = (char *)malloc(max_len);
254 if (temp) {
255 snprintf(temp, max_len, "(%s*%s)", buffer, current_clause);
256 strncpy(buffer, temp, max_len - 1);
257 buffer[max_len - 1] = '\0';
258 free(temp);
259 }
260 } else {
261 fclose(fp);
262 return 0;
263 }
264 }
265 current_clause[0] = '\0';
266 lits_in_clause = 0;
267 }
268 } else {
269 char lit_str[64];

271 snprintf(lit_str, sizeof(lit_str), "(~x%d)", -lit);
272 } else {
273 snprintf(lit_str, sizeof(lit_str), "x%d", lit);
274 }
275
276 if (lits_in_clause == 0) {
277 strncpy(current_clause, lit_str, sizeof(current_clause) - 1);
278 current_clause[sizeof(current_clause) - 1] = '\0';
279 } else {
280 char temp[8192];
281 snprintf(temp, sizeof(temp), "(%s+%s)", current_clause, lit_str);
282 strncpy(current_clause, temp, sizeof(current_clause) - 1);
283 current_clause[sizeof(current_clause) - 1] = '\0';
284 }

287 token = strtok(NULL, " \t\r\n");
288 }

292 return 0;
